import io from 'socket.io-client';
import { SERVER_URL } from '../config';

let socket;

export const initSocket = () => {
  socket = io(SERVER_URL, {
    transports: ['websocket'],
    reconnection: true,
    reconnectionDelay: 1000,
  });

  socket.on('connect', () => {
    console.log('✅ Socket connecté');
    socket.emit('subscribe:line', ['DDD 42', 'DDD 7']);
  });

  socket.on('gps:update', (data) => {
    // Met à jour la position du bus sur la carte
    store.dispatch(updateBusPosition(data));
  });

  socket.on('alert:passage', (alert) => {
    scheduleLocalNotification(alert);
  });
};

export default function MapScreen() {

  const route = useSelector(s => s.route.active);

  return (
    <MapView
      provider="google"
      initialRegion={{
        latitude: 14.6937, longitude: -17.4441,
        latitudeDelta: 0.05, longitudeDelta: 0.05,
      }}>
      {buses.map(bus => (
        <Marker
          key={bus._id}
          coordinate={bus.position}
          title={bus.line}
          description={`ETA: ${bus.etaMin}min`}
        />
      ))}
      {route && (
        <Polyline
          coordinates={route.coords}
          strokeColor="#22D3EE"
          strokeWidth={3}
        />
      )}
    </MapView>
  );
}

const Bus = require('../models/Bus');
const { io } = require('../server');
const { checkAlerts } = require('./fcmService');

// Polling GPS toutes les 5 secondes
setInterval(async () => {
  const buses = await Bus.find({ active: true });

  for (const bus of buses) {
    const gpsData = await fetchTrackerAPI(bus.trackerId);

    // Mise à jour MongoDB
    await Bus.updateOne({ _id: bus._id }, {
      position: gpsData.coords,
      speed: gpsData.speed,
      updatedAt: new Date(),
    });

    // Broadcast Socket.io par ligne
    io.to(`line:${bus.line}`).emit('gps:update', {
      busId: bus._id, line: bus.line,
      position: gpsData.coords,
      speed: gpsData.speed,
    });

    // Vérifier alertes de passage
    await checkAlerts(bus, gpsData);
  }
}, 5000);
const admin = require('firebase-admin');
const User = require('../models/User');

exports.checkAlerts = async (bus, gps) => {
  const stops = bus.upcomingStops;

  for (const stop of stops) {
    const dist = haversine(gps.coords, stop.coords);
    const etaMin = Math.round(dist / (bus.speed / 60));

    // Trouver utilisateurs abonnés à cet arrêt
    const users = await User.find({
      'alerts.stop': stop._id,
      'alerts.threshold': { $gte: etaMin },
    });

    if (users.length > 0) {
      await admin.messaging().sendMulticast({
        tokens: users.map(u => u.fcmToken),
        notification: {
          title: `${bus.line} dans ${etaMin} min`,
          body: `Arrêt ${stop.name} — Montez vite !`,
        },
      });
    }
  }
};
